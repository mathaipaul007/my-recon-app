/*******************************************************************************
  SECTION 1: DATABASE QUERIES (File 1)
*******************************************************************************/

-- Retrieve workflow configuration
SELECT * FROM recon_workflow WHERE id = '7';

-- Retrieve rule scripts and execution sequence
SELECT * FROM recon_rule_workflow rw
JOIN recon_rule_script_cfg rc ON rw.ruleid = rc.id
WHERE workflowid = 7
ORDER BY run_seq;
