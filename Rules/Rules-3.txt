-- SQL Queries for Workflow Configuration
select * from recon_workflow where id='3'

select * from recon_rule_workflow rw
join recon_rule_script_cfg rc
on rw.ruleid=rc.id
where workflowid=3
order by run_seq
# Rule #3 PREMIER-NB-Advice-Amount

PREMIER-SUM-AMOUNT
    FromBMOFundingFile().
    find("settlement_no", sum_("amounts").as_("sum_amt")).
    grouped_by("settlement_no")

COMPARE-SUM-INPUT-PREMIER
    FromInputStatement().
    find(col("id").as_("bmoinputid"),
    const(input["settlement_no"]).as_("sr_no")).
    where(jcol("description").contains("%0055%")).
    and_(jnum("net").rounded_abs_eq(input["sum_amt"],2))

GET-PREMIER-AMOUNT-BY-SRNO
    FromBMOFundingFile().
    find(col("id").as_("bmofundid"),
    jnum("amounts").as_("bmoamount"),
    const(input["bmoinputid"]).as_("bmoinputid"),
    const(input["sr_no"]).as_("sr_no")).
    where(jcol("settlement_no").is_(input["sr_no"])).
    and_(jcol("details").is_("BMO -Advices")).
    and_(col("cleared_status").is_("U"))

NB-SUM-AMOUNT
    FromNBFundingFile().
    find("settlement_no", sum_("amounts").as_("sum_amt"),
    const(input["bmofundid"]).as_("bmofundid"),
    const(input["bmoamount"]).as_("bmoamount"),
    const(input["bmoinputid"]).as_("bmoinputid")).
    where(jcol("settlement_no").is_(input["sr_no"])).
    grouped_by("settlement_no")

COMPARE-SUM-INPUT-NB
    FromInputStatement().
    find(UploadRawData.id.label("nbinputid"),
    const(input["settlement_no"]).label("sr_no"),
    const(input["bmofundid"]).label("bmofundid"),
    const(input["bmoamount"]).label("bmoamount"),
    const(input["bmoinputid"]).label("bmoinputid")).
    where(jcol("description").contains("%3677%")).
    and_(jnum("net").rounded_abs_eq(input["sum_amt"],2))

GET-NB-AMOUNT-BY-SRNO
    FromNBFundingFile().
    find(col("id").as_("nbfundid"),
    add_(UploadRawData.json_data,"amounts",literal(input["bmoamount"])).label("bmo-nbamounts"),
    const(input["bmoinputid"]).as_("bmoinputid"),
    const(input["bmofundid"]).as_("bmofundid"),
    const(input["nbinputid"]).as_("nbinputid")).
    where(jcol("settlement_no").is_(input["sr_no"])).
    and_(jcol("details").is_("NBS - Advices")).
    and_(col("cleared_status").is_("U"))

COMPARE-NB-PREMIER-AMOUT-SETTLEMENT
    FromInputStatement().
    find(col("id").as_("settlementid"),
    const(input["bmofundid"]).as_("bmofundid"),
    const(input["bmoinputid"]).as_("bmoinputid"),
    const(input["nbfundid"]).as_("nbfundid"),
    const(input["nbinputid"]).as_("nbinputid")).
    where(jcol("description").contains("%SETTLE%")).
    and_(col("cleared_status").is_("U")).
    and_(jnum("net").
    rounded_abs_eq(input["bmo-nbamounts"],2))

PREMIER-Amount-Report
    with_aliases(
        UploadRawData, "bmodata", "bmofunding",
        lambda bmodata, bmofunding:
        Q(bmodata)
        .join(bmofunding, on=bmofunding.id == input["bmofundid"])
        .find(func.replace(func.concat(bmodata.json_data["description"].astext, bmodata.json_data["net"].astext),'','').label("DESCRIPTION"),
        const("Funding").label("CATEGORY"),
        bmofunding.json_data["settlement_no"].astext.label("RUNID"),
        bmofunding.json_data["details"].astext.label("TYPE"),
        func.round(pg_safe_numeric(bmofunding.json_data, "amounts"),2).label("AMOUNT"),
        bmodata.json_data["value_date"].astext.label("Value_Date"))
        .where(bmodata.id == input["bmoinputid"])
    )

PREMIER-Settlement-Report
    FromInputStatement().
    find(UploadRawData.json_data["description"].astext.label("DESCRIPTION"),
    func.round(pg_safe_numeric(UploadRawData.json_data, "net"),2).label("AMOUNT"),
    UploadRawData.json_data["value_date"].astext.label("Value_Date"),
    literal("Advice").label("CATEGORY"))
    where(UploadRawData.id == input["settlementid"])

NB-Report
    with_aliases(
        UploadRawData, "nbdata", "nbfunding",
        lambda nbdata, nbfunding:
        Q(nbdata)
        .join(nbfunding, on=nbfunding.id == input["nbfundid"])
        .find(func.replace(func.concat(nbdata.json_data["description"].astext, nbdata.json_data["net"].astext),'','').label("DESCRIPTION"),
        literal("Funding").label("CATEGORY"),
        nbfunding.json_data["settlement_no"].astext.label("RUNID"),
        nbfunding.json_data["details"].astext.label("TYPE"),
        func.round(pg_safe_numeric(nbfunding.json_data, "amounts"),2).label("AMOUNT"),
        nbdata.json_data["value_date"].astext.label("Value_Date"))
        .where(nbdata.id == input["nbinputid"])
    )
